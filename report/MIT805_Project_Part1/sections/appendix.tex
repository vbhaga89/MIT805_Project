\subsection*{Dataset Details}

\begin{table}[H]
\centering
\caption{Description of variables in the \texttt{LI\_Large\_Trans.csv} dataset}
\label{tab:li_large_columns}
\renewcommand{\arraystretch}{1.15}
    \begin{tabular}{p{3.0cm} p{2.2cm} p{8.2cm}}
    \toprule
    \textbf{Variable} & \textbf{Data Type} & \textbf{Description} \\
    \midrule
    
    \texttt{Timestamp}
    & Datetime
    & Date and time at which the transaction occurred. \\
    \midrule
    
    \texttt{From Bank}
    & Integer
    & Identifier of the bank from which the transaction originated. \\
    \midrule
    
    \texttt{From Account}
    & String
    & Identifier of the account sending the funds. \\
    \midrule
    
    \texttt{To Bank}
    & Integer
    & Identifier of the bank receiving the transaction. \\
    \midrule
    
    \texttt{To Account}
    & String
    & Identifier of the account receiving the funds. \\
    \midrule
    
    \texttt{Amount Received}
    & Float
    & Monetary amount received by the destination account. \\
    \midrule
    
    \texttt{Receiving Currency}
    & Categorical
    & Currency in which the destination account received the funds. \\
    \midrule
    
    \texttt{Amount Paid}
    & Float
    & Monetary amount paid by the originating account. \\
    \midrule
    
    \texttt{Payment Currency}
    & Categorical
    & Currency in which the originating account made the payment. \\
    \midrule
    
    \texttt{Payment Format}
    & Categorical
    & Payment mechanism used for the transaction, such as cheque, credit card, ACH, or wire transfer. \\
    \midrule
    
    \texttt{Is Laundering}
    & Binary
    & Target variable indicating whether the transaction is associated with money laundering, where 1 denotes laundering and 0 denotes a legitimate transaction. \\
    
    \bottomrule
    \end{tabular}
\end{table}
